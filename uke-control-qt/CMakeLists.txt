cmake_minimum_required(VERSION 3.22)
project(uke-control-qt VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.6 REQUIRED COMPONENTS Widgets Qml Quick QuickControls2)
find_package(ECM 6.6 REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(ECMFindQmlModule)
ecm_find_qmlmodule(org.kde.kirigami 2.20 REQUIRED)
find_package(KF6KirigamiAddons 1.1 REQUIRED)
ecm_find_qmlmodule(org.kde.kirigamiaddons.formcard REQUIRED)
ecm_find_qmlmodule(org.kde.desktop REQUIRED)
find_package(KF6WidgetsAddons 6.6 REQUIRED)
find_package(KF6IconThemes 6.6 REQUIRED)
find_package(KF6ConfigWidgets 6.6 REQUIRED)
find_package(KF6I18n 6.8 REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
set(UKE_CONTROL_SOURCE "" CACHE PATH "Use this Uke Control source SDK instead of the installed SDK")
if(UKE_CONTROL_SOURCE)
  foreach(header client.hpp frontend.hpp)
    if(NOT EXISTS "${UKE_CONTROL_SOURCE}/src/${header}")
      message(FATAL_ERROR "UKE_CONTROL_SOURCE is missing src/${header}")
    endif()
  endforeach()
  find_library(UKE_SOURCE_LIBRARY NAMES uke-control-client libuke-control-client.so.1 PATHS "${UKE_CONTROL_SOURCE}/build" NO_DEFAULT_PATH NO_CACHE REQUIRED)
  add_library(uke-client SHARED IMPORTED)
  set_target_properties(uke-client PROPERTIES IMPORTED_LOCATION "${UKE_SOURCE_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${UKE_CONTROL_SOURCE}/src" INTERFACE_COMPILE_DEFINITIONS UKE_CONTROL_SOURCE_HEADERS=1)
  set(UKE_CLIENT uke-client)
else()
  pkg_check_modules(UKE QUIET IMPORTED_TARGET uke-control-client)
endif()
if(NOT UKE_CONTROL_SOURCE AND TARGET PkgConfig::UKE)
  set(UKE_CLIENT PkgConfig::UKE)
elseif(NOT UKE_CONTROL_SOURCE)
  find_library(UKE_LIBRARY NAMES uke-control-client libuke-control-client.so.1 HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../uke-control/build" REQUIRED)
  add_library(uke-client SHARED IMPORTED)
  set_target_properties(uke-client PROPERTIES IMPORTED_LOCATION "${UKE_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/../uke-control/src" INTERFACE_COMPILE_DEFINITIONS UKE_CONTROL_SOURCE_HEADERS=1)
  set(UKE_CLIENT uke-client)
endif()
add_library(uke-qt-model STATIC src/model-data.cpp src/model-data.hpp src/bridge.cpp src/bridge.hpp)
target_link_libraries(uke-qt-model PUBLIC Qt6::Core ${UKE_CLIENT} Threads::Threads)
target_compile_options(uke-qt-model PRIVATE -Wall -Wextra -Wpedantic)
add_executable(uke-control-qt src/main.cpp)
target_link_libraries(uke-control-qt PRIVATE uke-qt-model Qt6::Widgets Qt6::Quick Qt6::Qml Qt6::QuickControls2 KF6::IconThemes KF6::ConfigWidgets KF6::I18n KF6::I18nQml)
qt_policy(SET QTP0001 NEW)
set(qml_files Main.qml AreaPage.qml GuidePage.qml ObservationView.qml ReviewSheet.qml SafeText.qml)
list(TRANSFORM qml_files PREPEND "src/qml/")
foreach(qml_file IN LISTS qml_files)
  get_filename_component(qml_name "${qml_file}" NAME)
  set_source_files_properties("${qml_file}" PROPERTIES QT_RESOURCE_ALIAS "${qml_name}")
endforeach()
qt_add_qml_module(uke-control-qt URI org.uke.ControlQt VERSION 1.0 QML_FILES ${qml_files})
install(TARGETS uke-control-qt RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/org.uke.ControlQt.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/org.uke.ControlQt.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  # Keep the previous native view regression tests during migration. The shipped
  # executable uses the Kirigami frontend and does not link this widget library.
  add_library(uke-qt-ui STATIC src/ui.cpp src/ui.hpp)
  target_link_libraries(uke-qt-ui PUBLIC uke-qt-model Qt6::Widgets KF6::WidgetsAddons KF6::IconThemes KF6::ConfigWidgets)
  add_executable(uke-control-qt-tests tests/smoke.cpp)
  target_link_libraries(uke-control-qt-tests PRIVATE uke-qt-ui Qt6::Test)
  target_compile_definitions(uke-control-qt-tests PRIVATE FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../uke-control/tests/fixtures/tui.json")
  add_test(NAME qt-fixture-smoke COMMAND uke-control-qt-tests)
  set_tests_properties(qt-fixture-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME qt-cli-policy COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli-smoke.sh" "$<TARGET_FILE:uke-control-qt>")
  add_executable(uke-control-qt-appearance-tests tests/appearance.cpp)
  target_link_libraries(uke-control-qt-appearance-tests PRIVATE Qt6::Test Qt6::Gui)
  add_dependencies(uke-control-qt-appearance-tests uke-control-qt)
  add_test(NAME qt-native-appearance COMMAND uke-control-qt-appearance-tests)
  set_tests_properties(qt-native-appearance PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(uke-control-kirigami-tests tests/kirigami.cpp)
  qt_add_qml_module(uke-control-kirigami-tests URI org.uke.ControlQtTest VERSION 1.0 QML_FILES ${qml_files})
  target_link_libraries(uke-control-kirigami-tests PRIVATE uke-qt-model Qt6::Test Qt6::Widgets Qt6::Quick Qt6::Qml Qt6::QuickControls2 KF6::IconThemes KF6::ConfigWidgets KF6::I18n KF6::I18nQml)
  target_compile_definitions(uke-control-kirigami-tests PRIVATE FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../uke-control/tests/fixtures/tui.json")
  add_test(NAME qt-kirigami COMMAND uke-control-kirigami-tests)
  set_tests_properties(qt-kirigami PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
  set_tests_properties(qt-kirigami PROPERTIES TIMEOUT 90)
endif()
